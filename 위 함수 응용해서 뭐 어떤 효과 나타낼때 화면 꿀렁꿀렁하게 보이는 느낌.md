#include <windows.h>
#include <iostream>
#include <thread>
#include <chrono>

int main() {
    HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);

    // 콘솔 버퍼 크기 설정 (1000x1000 크기의 맵)
    COORD bufferSize = { 1000, 1000 };
    if (!SetConsoleScreenBufferSize(hConsole, bufferSize)) {
        std::cerr << "Failed to set console screen buffer size.\n";
        return 1;
    }

    // 화면에 표시되는 창 크기 설정 (20x10)
    SMALL_RECT windowSize = { 0, 0, 19, 9 };  // Left, Top, Right, Bottom (0-based index)
    if (!SetConsoleWindowInfo(hConsole, TRUE, &windowSize)) {
        std::cerr << "Failed to set console window size.\n";
        return 1;
    }

    // 간단한 맵을 출력 (예: 텍스트로 맵 생성)
    for (int y = 0; y < 1000; ++y) {
        for (int x = 0; x < 1000; ++x) {
            std::cout << '.';
        }
        std::cout << '\n';
    }

    // 맵을 이동하는 부분
    int mapOffsetX = 0;
    int mapOffsetY = 0;
    while (true) {
        // 맵을 이동
        windowSize.Left = mapOffsetX;
        windowSize.Top = mapOffsetY;
        windowSize.Right = mapOffsetX + 19;  // 창 크기 20
        windowSize.Bottom = mapOffsetY + 9;  // 창 크기 10

        // 윈도우 크기 설정 (맵 이동)
        if (!SetConsoleWindowInfo(hConsole, TRUE, &windowSize)) {
            std::cerr << "Failed to set console window size.\n";
            break;
        }

        // 맵이 일정 시간마다 이동
        mapOffsetX += 1;
        mapOffsetY += 1;

        // 딜레이를 줘서 스크롤 느낌을 줍니다.
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }

    return 0;
}
