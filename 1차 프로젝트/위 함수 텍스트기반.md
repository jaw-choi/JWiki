#include <iostream>
#include <chrono>
#include <thread>

void clearScreen() {
    // 콘솔 화면을 지우는 방법 (Windows에서 클리어)
    std::cout << "\033[2J\033[1;1H";  // ANSI escape sequence for clearing screen
}

int main() {
    int width = 20;  // 화면 너비
    int height = 10; // 화면 높이
    int maxWidth = 50;  // 최대 화면 너비
    int maxHeight = 20; // 최대 화면 높이

    // 효과 시작
    for (int i = 1; i <= 5; ++i) {
        // 화면 크기를 확대/축소
        width = i * 5;
        height = i * 2;

        // 화면 지우기
        clearScreen();

        // 새로운 크기로 텍스트를 출력
        for (int y = 0; y < height; ++y) {
            for (int x = 0; x < width; ++x) {
                std::cout << '.';
            }
            std::cout << '\n';
        }

        // 딜레이 추가 (애니메이션 효과)
        std::this_thread::sleep_for(std::chrono::milliseconds(300));  // 300ms 딜레이
    }

    return 0;
}
